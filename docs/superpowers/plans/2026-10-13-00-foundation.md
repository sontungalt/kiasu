## Global Constraints

- Timezone is **Asia/Singapore (UTC+8, no DST)**. **Every stored timestamp is a UTC ISO string** produced by `Date.prototype.toISOString()` (ends in `Z`). Convert at the boundaries with `toUtcIso()`.
- Triage model `claude-haiku-5-5`, chat model `claude-sonnet-5-5`. Both are env vars (`TRIAGE_MODEL`, `CHAT_MODEL`). Never hard-code them elsewhere.
- Gmail access is **read-only**. No code path may send, delete, modify or forward email.
- Only WhatsApp messages from `MY_WHATSAPP_NUMBER` (digits only, set in the gitignored `.env`) are processed.
- Never log email bodies. Log IDs, subjects, counts and error codes.
- Never commit secrets, `.env`, or real email fixtures (`evals/triage/fixtures/`, `evals/triage/labels.json` are gitignored).
- Relative imports use the `.js` suffix (NodeNext ESM), e.g. `import { x } from "./clock.js"`.
- Tests run with `npm test` (vitest). Every task ends with `npm test` and `npm run typecheck` passing.

## Review Focus

1. **Timestamps with offsets.** If a stored `dueAt` were `+08:00` and another value `Z`, string comparison would sort them wrongly. `toUtcIso` must be applied before every store write. *Pinned by:* the Task 3 test `toUtcIso normalises +08:00`, and the Task 7 test `upsert stores UTC`.
2. **SGT day boundaries.** 23:30 UTC is already the next day in SGT, so "today" must be computed in SGT, not UTC. *Pinned by:* the Task 3 test `sgtDateKey rolls over at 16:00Z`.
3. **Quiet hours that wrap midnight** (22:30–06:30). Both 23:00 and 05:00 must count as quiet. *Pinned by:* the Task 3 test `quiet hours wrap midnight`.
4. **Duplicate deliveries.** Pub/Sub and Meta retry, so `processed.claim` must return `true` exactly once per key, including under the Firestore implementation. *Pinned by:* the Task 5 contract test `claim is once-only`, which runs against both stores.
5. **Follow-up emails duplicating items.** "Reminder: form due tomorrow" in the same thread must update the existing item, not create a new one. *Pinned by:* the Task 7 test `same thread + similar title updates instead of creating`.

```ts
import { z } from "zod";

const bool = z.enum(["true", "false"]).default("false").transform((v) => v === "true");

const EnvSchema = z
  .object({
    NODE_ENV: z.enum(["development", "test", "production"]).default("development"),
    PORT: z.coerce.number().int().default(8080),
    GCP_PROJECT: z.string().min(1),
    GCP_REGION: z.string().default("asia-southeast1"),
    PUBLIC_BASE_URL: z.url(),
    ANTHROPIC_API_KEY: z.string().min(1),
    TRIAGE_MODEL: z.string().default("claude-haiku-5-5"),
    CHAT_MODEL: z.string().default("claude-sonnet-5-5"),
    GOOGLE_OAUTH_CLIENT_ID: z.string().min(1),
    GOOGLE_OAUTH_CLIENT_SECRET: z.string().min(1),
    GOOGLE_REFRESH_TOKEN_PERSONAL: z.string().optional(),
    GOOGLE_REFRESH_TOKEN_SCHOOL: z.string().optional(),
    GMAIL_PUBSUB_TOPIC: z.string().min(1),
    INVOKER_SERVICE_ACCOUNT: z.email(),
    TASKS_QUEUE: z.string().min(1),
    WA_PHONE_NUMBER_ID: z.string().min(1),
    WA_ACCESS_TOKEN: z.string().min(1),
    WA_APP_SECRET: z.string().min(1),
    WA_VERIFY_TOKEN: z.string().min(1),
    WA_GRAPH_VERSION: z.string().default("v23.0"),
    MY_WHATSAPP_NUMBER: z.string().regex(/^\d{8,15}$/, "digits only, with country code, e.g. 6512345678"),
    CLASSROOM_ENABLED: z.enum(["true", "false"]).default("true").transform((v) => v === "true"),
    DEV_SKIP_OIDC: bool,
  })
  .refine((c) => !(c.NODE_ENV === "production" && c.DEV_SKIP_OIDC), {
    message: "DEV_SKIP_OIDC must not be true in production",
    path: ["DEV_SKIP_OIDC"],
  });

export type Config = z.infer<typeof EnvSchema>;

export function loadConfig(env: Record<string, string | undefined> = process.env): Config {
  const result = EnvSchema.safeParse(env);
  if (!result.success) {
    throw new Error(`Invalid configuration:\n${z.prettifyError(result.error)}`);
  }
  return result.data;
}
```

`src/lib/clock.ts`:
```ts
export interface Clock {
  now(): Date;
}

export const HOUR_MS = 3_600_000;
const SGT_OFFSET_MS = 8 * HOUR_MS; // Singapore has no DST
const WEEKDAYS = ["Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"];
const MONTHS = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];

export const systemClock: Clock = { now: () => new Date() };

export class FakeClock implements Clock {
  private t: number;
  constructor(iso: string) {
    this.t = parse(iso).getTime();
  }
  now(): Date {
    return new Date(this.t);
  }
  set(iso: string): void {
    this.t = parse(iso).getTime();
  }
  advance(ms: number): void {
    this.t += ms;
  }
}

function parse(input: string | Date): Date {
  const d = input instanceof Date ? new Date(input.getTime()) : new Date(input);
  if (Number.isNaN(d.getTime())) throw new Error(`Invalid date: ${String(input)}`);
  return d;
}

/** Shift so that the UTC getters read Singapore wall-clock values. */
function shifted(d: Date): Date {
  return new Date(d.getTime() + SGT_OFFSET_MS);
}

export function toUtcIso(input: string | Date): string {
  return parse(input).toISOString();
}

export function sgtDateKey(d: Date): string {
  return shifted(d).toISOString().slice(0, 10);
}

export function sgtTimeHHMM(d: Date): string {
  return shifted(d).toISOString().slice(11, 16);
}

export function sgtAt(dateKey: string, hhmm: string): Date {
  return parse(`${dateKey}T${hhmm}:00+08:00`);
}

export function addDaysToKey(dateKey: string, days: number): string {
  const d = new Date(`${dateKey}T00:00:00Z`);
  d.setUTCDate(d.getUTCDate() + days);
  return d.toISOString().slice(0, 10);
}

export function startOfSgtDay(d: Date): Date {
  return sgtAt(sgtDateKey(d), "00:00");
}

export function formatSgt(d: Date, style: "time" | "dayDate" | "dayDateTime"): string {
  const s = shifted(d);
  const h24 = s.getUTCHours();
  const m = s.getUTCMinutes();
  const h12 = h24 % 12 === 0 ? 12 : h24 % 12;
  const ampm = h24 < 12 ? "am" : "pm";
  const time = m === 0 ? `${h12}${ampm}` : `${h12}:${String(m).padStart(2, "0")}${ampm}`;
  const dayDate = `${WEEKDAYS[s.getUTCDay()]} ${s.getUTCDate()} ${MONTHS[s.getUTCMonth()]}`;
  if (style === "time") return time;
  if (style === "dayDate") return dayDate;
  return `${dayDate}, ${time}`;
}

function minutesOf(hhmm: string): number {
  const [h, m] = hhmm.split(":").map(Number);
  return (h ?? 0) * 60 + (m ?? 0);
}

/** True if d (SGT wall clock) is in [quietStart, quietEnd). Handles windows that wrap midnight. */
export function isQuietTime(d: Date, quietStart: string, quietEnd: string): boolean {
  const now = minutesOf(sgtTimeHHMM(d));
  const start = minutesOf(quietStart);
  const end = minutesOf(quietEnd);
  if (start === end) return false;
  return start < end ? now >= start && now < end : now >= start || now < end;
}
```

```ts
export type LogFields = Record<string, unknown>;

export interface Logger {
  info(message: string, fields?: LogFields): void;
  warn(message: string, fields?: LogFields): void;
  error(message: string, fields?: LogFields): void;
  child(fields: LogFields): Logger;
}

function serialise(fields: LogFields): LogFields {
  const out: LogFields = {};
  for (const [k, v] of Object.entries(fields)) {
    out[k] = v instanceof Error ? { message: v.message, name: v.name, stack: v.stack } : v;
  }
  return out;
}

/** Cloud Logging reads `severity` and `message` from JSON lines on stdout. */
export function createLogger(base: LogFields = {}, write: (line: string) => void = (l) => console.log(l)): Logger {
  const emit = (severity: string) => (message: string, fields: LogFields = {}) =>
    write(JSON.stringify({ severity, message, ...serialise(base), ...serialise(fields) }));
  return {
    info: emit("INFO"),
    warn: emit("WARNING"),
    error: emit("ERROR"),
    child: (fields) => createLogger({ ...base, ...fields }, write),
  };
}

export const silentLogger: Logger = createLogger({}, () => {});
```

```ts
import type { Item } from "../types.js";
import type { ItemFilter } from "./types.js";

export function itemAccount(item: Item): string | null {
  if (item.source.kind === "email") return item.source.account;
  if (item.source.kind === "classroom") return "school";
  return null;
}

export function matchesFilter(item: Item, f: ItemFilter): boolean {
  if (f.status && !f.status.includes(item.status)) return false;
  if (f.hasDue === true && !item.dueAt) return false;
  if (f.hasDue === false && item.dueAt) return false;
  if (f.dueBefore && !(item.dueAt && item.dueAt < f.dueBefore)) return false;
  if (f.dueAfter && !(item.dueAt && item.dueAt >= f.dueAfter)) return false;
  if (f.account && itemAccount(item) !== f.account) return false;
  if (f.sourceKind && item.source.kind !== f.sourceKind) return false;
  if (f.query) {
    const q = f.query.toLowerCase();
    if (!`${item.title} ${item.details ?? ""}`.toLowerCase().includes(q)) return false;
  }
  return true;
}

/** Due items first (soonest first), then undated by createdAt. */
export function sortItems(items: Item[]): Item[] {
  return [...items].sort((a, b) => {
    if (a.dueAt && b.dueAt) return a.dueAt.localeCompare(b.dueAt);
    if (a.dueAt) return -1;
    if (b.dueAt) return 1;
    return a.createdAt.localeCompare(b.createdAt);
  });
}

export function applyFilter(items: Item[], f: ItemFilter): Item[] {
  const out = sortItems(items.filter((i) => matchesFilter(i, f)));
  return f.limit ? out.slice(0, f.limit) : out;
}
```

```ts
    listPending: async (limit) =>

      if (!cur) throw new Error(`Reminder not found: ${id}`);

      clone([...this.reminderMap.values()].filter((r) => r.status === "pending" && r.fireAt <= now)

        .sort((a, b) => a.fireAt.localeCompare(b.fireAt))),

    append: async (t) => void this.turns.push({ ...clone(t), id: randomUUID() }),
    recent: async (n) => clone([...this.turns].sort((a, b) => a.at.localeCompare(b.at)).slice(-n)),

    create: async (data) => {
      const m: OutboxMessage = { ...clone(data), id: randomUUID() };
      this.outboxMap.set(m.id, m);
      return clone(m);
    },
    update: async (id, patch) => {
      const cur = this.outboxMap.get(id);
      if (!cur) throw new Error(`Outbox message not found: ${id}`);
      this.outboxMap.set(id, { ...cur, ...clone(patch) });
    },
    listPending: async () =>
      clone([...this.outboxMap.values()].filter((m) => m.status === "pending")
        .sort((a, b) => a.createdAt.localeCompare(b.createdAt))),
  };

  state: Store["state"] = {
    get: async <K extends keyof StateDocs>(key: K) => clone((this.stateMap.get(key) as StateDocs[K] | undefined) ?? null),
    set: async <K extends keyof StateDocs>(key: K, value: StateDocs[K]) => void this.stateMap.set(key, clone(value)),
  };

  settings: Store["settings"] = {
    get: async () => ({ ...DEFAULT_SETTINGS, ...clone(this.settingsOverride) }),
  };

  processed: Store["processed"] = {
    claim: async (key, at) => {
      if (this.processedMap.has(key)) return false;
      this.processedMap.set(key, at);
      return true;

    purgeOlderThan: async (iso) => {
      let n = 0;
      for (const [k, at] of this.processedMap) {
        if (at < iso) {
          this.processedMap.delete(k);
          n++;
        }
      }
      return n;
    },
  };
```

```ts
  AccountState, ConversationTurn, EmailRecord, Item, OutboxMessage, Reminder, Settings, StateDocs,

  return new Firestore({ projectId, ignoreUndefinedProperties: true });

const docId = (key: string) => key.replace(/\//g, "_");

const ALREADY_EXISTS = 6;

export class FirestoreStore implements Store {
  constructor(private db: Firestore) {}

  private col(name: string) {
    return this.db.collection(name);
  }

  private async all<T>(q: FirebaseFirestore.Query): Promise<T[]> {
    const snap = await q.get();
    return snap.docs.map((d) => ({ ...(d.data() as DocumentData), id: d.id }) as T);
  }

  private async one<T>(name: string, id: string): Promise<T | null> {
    const snap = await this.col(name).doc(docId(id)).get();
    return snap.exists ? ({ ...(snap.data() as DocumentData), id: snap.id } as T) : null;
  }

  accounts: Store["accounts"] = {
    get: (id) => this.one<AccountState>("accounts", id),
    upsert: async (a) => void (await this.col("accounts").doc(a.id).set(a)),
  };

  emails: Store["emails"] = {
    get: (id) => this.one<EmailRecord>("emails", id),
    upsert: async (e) => void (await this.col("emails").doc(docId(e.id)).set(e)),
    listPending: async (limit) =>
      (await this.all<EmailRecord>(this.col("emails").where("triageStatus", "==", "pending")))
        .sort((a, b) => a.receivedAt.localeCompare(b.receivedAt)).slice(0, limit),
    listImportantSince: async (since) =>
      (await this.all<EmailRecord>(this.col("emails").where("processedAt", ">=", since)))
        .filter((e) => e.triageStatus === "done" && (e.importance === "high" || e.importance === "medium"))
        .sort((a, b) => b.receivedAt.localeCompare(a.receivedAt)),
  };

  items: Store["items"] = {
    get: (id) => this.one<Item>("items", id),
    create: async (data) => {
      const ref = this.col("items").doc();
      await ref.set(data);
      return { ...data, id: ref.id };
    },
    update: async (id, patch) => void (await this.col("items").doc(id).update(patch as DocumentData)),
    findByDedupeKey: async (key) =>
      (await this.all<Item>(this.col("items").where("dedupeKey", "==", key).limit(1)))[0] ?? null,
    listByThread: (threadId) => this.all<Item>(this.col("items").where("source.threadId", "==", threadId)),
    list: async (filter: ItemFilter) => {
      const q = filter.status && filter.status.length > 0

        : this.col("items");
      return applyFilter(await this.all<Item>(q), filter);

    create: async (data) => {
      const ref = this.col("reminders").doc();

    listDue: async (now) =>

      const snap = await this.col("state").doc(key).get();

    get: async () => {
      const snap = await this.col("settings").doc("main").get();
      return { ...DEFAULT_SETTINGS, ...((snap.data() as Partial<Settings> | undefined) ?? {}) };
    },
  };

  processed: Store["processed"] = {
    claim: async (key, at) => {
      try {

        return true;
      } catch (err) {
        if ((err as { code?: number }).code === ALREADY_EXISTS) return false;
        throw err;
      }

      snap.docs.forEach((d) => batch.delete(d.ref));
      if (!snap.empty) await batch.commit();
      return snap.size;
    },
  };
```

```ts
import type { Store } from "../store/types.js";
import type { Item, ItemSource, ItemType, Urgency } from "../types.js";
import { toUtcIso } from "../lib/clock.js";

export interface ItemDraft {
  type: ItemType;
  title: string;
  details: string | null;
  dueAt: string | null;
  startAt: string | null;
  endAt: string | null;
  urgency: Urgency;
  urgencyReason: string | null;
  source: ItemSource;
  dedupeKey: string;
}

export interface UpsertResult {
  item: Item;
  change: "created" | "updated" | "unchanged";
  dueChanged: boolean;
}

const URGENCY_RANK: Record<Urgency, number> = { low: 0, normal: 1, urgent: 2 };
const STOPWORDS = new Set(["the", "a", "an", "to", "for", "of", "your", "my", "by", "on", "please", "tomorrow", "today"]);

export function normaliseTitle(title: string): string {
  let t = title.toLowerCase().trim();
  // strip repeated "re:", "fwd:", "fw:", "reminder:" prefixes
  for (;;) {
    const next = t.replace(/^(re|fwd?|reminder)\s*:\s*/, "");
    if (next === t) break;
    t = next;
  }
  return t.replace(/[^a-z0-9\s]/g, " ").replace(/\s+/g, " ").trim();
}

export function emailDedupeKey(threadId: string, title: string): string {
  return `${threadId}:${normaliseTitle(title)}`;
}

function tokens(title: string): Set<string> {
  return new Set(normaliseTitle(title).split(" ").filter((w) => w && !STOPWORDS.has(w)));
}

function similarity(a: string, b: string): number {
  const ta = tokens(a);
  const tb = tokens(b);
  if (ta.size === 0 || tb.size === 0) return 0;
  let inter = 0;
  for (const w of ta) if (tb.has(w)) inter++;
  return inter / Math.min(ta.size, tb.size);
}

const iso = (v: string | null) => (v ? toUtcIso(v) : null);

async function findExisting(store: Store, draft: ItemDraft, dueAt: string | null): Promise<Item | null> {
  const byKey = await store.items.findByDedupeKey(draft.dedupeKey);
  if (byKey) return byKey;
  if (draft.source.kind !== "email") return null;
  const siblings = await store.items.listByThread(draft.source.threadId);
  return (
    siblings.find((s) => s.type === draft.type && similarity(s.title, draft.title) >= 0.6) ??
    siblings.find((s) => s.type === draft.type && dueAt !== null && s.dueAt === dueAt) ??
    null
  );
}

export async function upsertItem(store: Store, draft: ItemDraft, now: Date): Promise<UpsertResult> {
  const nowIso = now.toISOString();
  const dueAt = iso(draft.dueAt);
  const startAt = iso(draft.startAt);
  const endAt = iso(draft.endAt);
  const existing = await findExisting(store, draft, dueAt);

  if (!existing) {
    const item = await store.items.create({
      type: draft.type, title: draft.title, details: draft.details,
      dueAt, startAt, endAt, urgency: draft.urgency, urgencyReason: draft.urgencyReason,
      status: "open", snoozedUntil: null, nudgedAt: null, alertedAt: null,
      source: draft.source, dedupeKey: draft.dedupeKey, createdAt: nowIso, updatedAt: nowIso,
    });
    return { item, change: "created", dueChanged: false };
  }

  const patch: Partial<Item> = {};
  const dueChanged = dueAt !== null && dueAt !== existing.dueAt;
  if (dueChanged) {
    patch.dueAt = dueAt;
    patch.alertedAt = null;
    patch.nudgedAt = null;
  }
  if (startAt !== null && startAt !== existing.startAt) patch.startAt = startAt;
  if (endAt !== null && endAt !== existing.endAt) patch.endAt = endAt;
  if (draft.details && draft.details !== existing.details) patch.details = draft.details;
  if (URGENCY_RANK[draft.urgency] > URGENCY_RANK[existing.urgency]) {
    patch.urgency = draft.urgency;
    patch.urgencyReason = draft.urgencyReason;
  }

  if (Object.keys(patch).length === 0) return { item: existing, change: "unchanged", dueChanged: false };
  patch.updatedAt = nowIso;
  await store.items.update(existing.id, patch);
  return { item: { ...existing, ...patch }, change: "updated", dueChanged };
}
```

```ts
import type { Importance, Settings } from "../types.js";
import type { UpsertResult } from "./upsert.js";
import { HOUR_MS } from "../lib/clock.js";

export interface AlertContext {
  importance: Importance;
  senderEmail: string | null;
}

export function senderEmailOf(from: string): string | null {
  const m = from.match(/<([^>]+)>/) ?? from.match(/([^\s<>]+@[^\s<>]+)/);
  return m?.[1] ? m[1].trim().toLowerCase() : null;
}

export function isVip(email: string, vipSenders: string[]): boolean {
  const e = email.toLowerCase();
  return vipSenders.some((v) => {
    const s = v.toLowerCase();
    return s.startsWith("@") ? e.endsWith(s) : e === s;
  });
}

export function shouldAlert(result: UpsertResult, ctx: AlertContext, settings: Settings, now: Date): boolean {
  const { item, change, dueChanged } = result;
  if (item.status !== "open") return false;
  if (change === "unchanged") return false;
  if (change === "updated" && !dueChanged) return false;
  if (item.alertedAt) return false;

  const dueMs = item.dueAt ? Date.parse(item.dueAt) : null;
  const dueSoon = dueMs !== null && dueMs > now.getTime() && dueMs - now.getTime() <= settings.urgentWindowHours * HOUR_MS;

  if (item.urgency === "urgent") return true;
  if (dueSoon && ctx.importance === "high") return true;
  if (dueChanged && dueSoon) return true;
  if (ctx.senderEmail && isVip(ctx.senderEmail, settings.vipSenders) && ctx.importance !== "low") return true;
  return false;
}
```

```ts
import { OAuth2Client } from "google-auth-library";
import type { preHandlerHookHandler } from "fastify";
import type { Deps } from "../ports.js";

export type IdTokenVerifier = (
  idToken: string,
  audience: string,
) => Promise<{ email?: string; email_verified?: boolean } | undefined>;

const client = new OAuth2Client();
const googleVerifier: IdTokenVerifier = async (idToken, audience) =>
  (await client.verifyIdToken({ idToken, audience })).getPayload();

export async function verifyGoogleOidc(
  authHeader: string | undefined,
  opts: { audience: string; email: string },
  verify: IdTokenVerifier = googleVerifier,
): Promise<boolean> {
  if (!authHeader?.startsWith("Bearer ")) return false;
  try {
    const payload = await verify(authHeader.slice("Bearer ".length), opts.audience);
    return !!payload && payload.email === opts.email && payload.email_verified === true;
  } catch {
    return false;
  }
}

export function oidcPreHandler(
  deps: Pick<Deps, "config" | "log">,
  verify: IdTokenVerifier = googleVerifier,
): preHandlerHookHandler {
  return async (request, reply) => {
    if (deps.config.DEV_SKIP_OIDC) return;
    const path = request.url.split("?")[0] ?? "";
    const audience = `${deps.config.PUBLIC_BASE_URL.replace(/\/$/, "")}${path}`;
    const ok = await verifyGoogleOidc(request.headers.authorization, { audience, email: deps.config.INVOKER_SERVICE_ACCOUNT }, verify);
    if (!ok) {
      deps.log.warn("oidc rejected", { path });
      await reply.code(401).send({ error: "unauthorized" });
    }
  };
}
```

```ts
import Fastify, { type FastifyInstance } from "fastify";
import type { Deps } from "./ports.js";

export type RouteRegistrar = (app: FastifyInstance, deps: Deps) => void | Promise<void>;

export async function buildServer(deps: Deps, registrars: RouteRegistrar[]): Promise<FastifyInstance> {
  const app = Fastify({ logger: false, bodyLimit: 1_048_576 });

  app.setErrorHandler((err, request, reply) => {
    deps.log.error("request failed", { err, path: request.url.split("?")[0] });
    const status = err.statusCode && err.statusCode < 500 ? err.statusCode : 500;
    void reply.code(status).send({ error: status === 500 ? "internal" : err.message });
  });

  app.get("/healthz", async () => ({ ok: true }));

  for (const register of registrars) {
    await app.register(async (scope) => {
      await register(scope, deps);
    });
  }
  await app.ready();
  return app;
}
```
